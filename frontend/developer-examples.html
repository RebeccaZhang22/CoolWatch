示例代码

把下面的检测函数放到已有智能体的模型调用前和工具返回后。请通过环境变量保存 Token，不要写入源码或提交到代码仓库。

Bash · 单次上下文检测
export PROSPECTMONITOR_API_KEY="YOUR_TOKEN"
export PROSPECTMONITOR_BASE_URL="http://localhost:18088"

curl --fail-with-body --silent --show-error \
  "$PROSPECTMONITOR_BASE_URL/v1/moderations" \
  -H "Authorization: Bearer $PROSPECTMONITOR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "messages": [
      {"role": "system", "content": "你是企业客服助手。"},
      {"role": "user", "content": "请查询我的订单状态。"}
    ],
    "tools": [
      {
        "type": "function",
        "function": {
          "name": "lookup_order",
          "description": "查询订单"
        }
      }
    ]
  }'
Python · 接入已有智能体
import os
import requests

BASE_URL = os.getenv("PROSPECTMONITOR_BASE_URL", "http://localhost:18088")
API_KEY = os.environ["PROSPECTMONITOR_API_KEY"]


def moderate(messages, tools=None):
    response = requests.post(
        f"{BASE_URL}/v1/moderations",
        headers={"Authorization": f"Bearer {API_KEY}"},
        json={"messages": messages, "tools": tools},
        timeout=65,
    )
    response.raise_for_status()
    return response.json()


def require_pass(messages, tools=None):
    result = moderate(messages, tools)
    if result["action"] == "block":
        raise RuntimeError(
            f"安全检测已拦截本轮执行，request_id={result['request_id']}"
        )
    return result


tools = [{
    "type": "function",
    "function": {
        "name": "lookup_order",
        "description": "查询订单",
        "parameters": {"type": "object", "properties": {}},
    },
}]
messages = [
    {"role": "system", "content": "你是企业客服助手。"},
    {"role": "user", "content": "请查询我的订单状态。"},
]

# 检测点 1：调用业务模型前
require_pass(messages, tools)
# assistant_message = your_existing_agent.run(messages, tools=tools)

# 工具执行完毕后，将模型的工具调用和真实结果加入现有上下文
messages.append({
    "role": "assistant",
    "content": "",
    "tool_calls": [{
        "id": "call_1",
        "type": "function",
        "function": {"name": "lookup_order", "arguments": "{}"},
    }],
})
messages.append({
    "role": "tool",
    "tool_call_id": "call_1",
    "content": '{"order_id":"A1024","status":"shipped"}',
})

# 检测点 2：把工具结果交回业务模型前
require_pass(messages, tools)
